      set: (k, v) => localStorage.setItem('ss_' + k, JSON.stringify(v)),
      clear: () => Object.keys(localStorage).filter(k => k.startsWith('ss_')).forEach(k => localStorage.removeItem(k))
    };

    const API = window.location.origin.includes('localhost') || window.location.origin.includes('127.0.0.1')
      ? (window.location.port === '5000' ? '/api/' : 'http://localhost:5000/api/')
      : 'http://localhost:5000/api/';

    async function apiFetch(path, method = 'GET', body = null) {
      const token = localStorage.getItem('ss_token');
      const opts = { method, headers: { 'Content-Type': 'application/json' } };
      if (token) opts.headers['Authorization'] = 'Bearer ' + token;
      if (body) opts.body = JSON.stringify(body);
      try {
        const res = await fetch(API + path, opts);
        if (!res.ok) {
          const err = await res.json().catch(() => ({}));
          return { success: false, error: err.error || 'Server error ' + res.status };
        }
        return await res.json();
      } catch (e) {
        return { success: false, error: 'Cannot reach server. Is Flask running?' };
      }
    }

    // ══════════ NOTIFICATIONS ══════════
    let notifications = [];

      const el = document.getElementById('notifList');
      if (!notifications.length) { el.innerHTML = '<div style="padding:24px;text-align:center;color:var(--ink3);font-size:13px;">No notifications</div>'; return; }
      el.innerHTML = notifications.map(n => `
    <div class="notif-item ${n.unread ? 'unread' : ''}" onclick="notifClick('${n.userId || ''}','${n.convoId || ''}')">
      <div class="notif-item-icon">${n.icon}</div>
      <div class="notif-item-body"><p>${n.text}</p><span>${n.time}</span></div>
    </div>`).join('');
    }
    function notifClick(userId, convoId) {
      document.getElementById('notifPanel').classList.remove('open');

    function toggleNotifPanel() {
      const p = document.getElementById('notifPanel');

        renderNotifList();

    async function loadMessages(convoId, silent = false) {
      if (convoId && convoId.startsWith('local_')) {
        const userId = convoId.replace('local_', '');
        const stored = JSON.parse(localStorage.getItem('ss_local_msgs_' + userId) || '[]');
        const w = document.getElementById('msgsWrap');
        if (!stored.length) {
          if (!silent) w.innerHTML = '<div style="padding:24px;text-align:center;color:var(--ink3);font-size:13px;">👋 Say hello!</div>';
          return;
        }
        w.innerHTML = stored.map(m => `<div class="msg ${m.mine ? 'sent' : 'recv'}"><div class="msg-bubble">${escHtml(m.text)}</div><div class="msg-time">${m.time} ✓</div></div>`).join('');
        w.scrollTop = w.scrollHeight;
        return;
      }
      const data = await apiFetch('conversations/' + convoId + '/messages');
      if (!data.success) { if (!silent) toast('Could not load messages', 'error'); return; }
      const w = document.getElementById('msgsWrap');
      const msgs = data.data || [];
      if (!msgs.length) {
        if (!silent) w.innerHTML = '<div style="padding:24px;text-align:center;color:var(--ink3);font-size:13px;">👋 Say hello!</div>';
        return;
      }
      const myId = S.user?.id;
      const atBottom = w.scrollHeight - w.scrollTop - w.clientHeight < 80;
      w.innerHTML = msgs.map(m => {
        const isMine = m.sender_id === myId || m.sender_id == myId;
        const time = m.created_at ? new Date(m.created_at).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }) : '';
        return `<div class="msg ${isMine ? 'sent' : 'recv'}"><div class="msg-bubble">${escHtml(m.content || m.text || '')}</div><div class="msg-time">${time}${isMine ? ' ✓' : ''}</div></div>`;
      }).join('');
      if (atBottom || !silent) w.scrollTop = w.scrollHeight;
    }

    function escHtml(str) { return str.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;'); }

    function buildCoach() { const subs = COACH_POOL.sort(() => Math.random() - .5).slice(0, 3); document.getElementById('coachRow').innerHTML = subs.map(s => `<div class="coach-chip" onclick="useCoach('${s.replace(/'/g, "&#39;")}')">${s}</div>`).join(''); }

    async function sendMsg() {
      const inp = document.getElementById('chatInput');
      const txt = inp.value.trim();
      if (!txt) return;
      inp.value = '';
      const w = document.getElementById('msgsWrap');
      const placeholder = w.querySelector('div[style]');
      if (placeholder && !placeholder.classList.contains('msg')) placeholder.remove();
      const now = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
      const msgDiv = document.createElement('div');
      msgDiv.className = 'msg sent';
      msgDiv.innerHTML = `<div class="msg-bubble">${escHtml(txt)}</div><div class="msg-time">${now} ✓</div>`;
      w.appendChild(msgDiv);
      w.scrollTop = w.scrollHeight;
      if (!S.activeChat || S.activeChat.startsWith('local_')) {
        const userId = S.activeChat ? S.activeChat.replace('local_', '') : 'unknown';
        const key = 'ss_local_msgs_' + userId;
        const stored = JSON.parse(localStorage.getItem(key) || '[]');
        stored.push({ text: txt, mine: true, time: now });
        localStorage.setItem(key, JSON.stringify(stored));
        renderLocalChatList();
        return;
      }
      const data = await apiFetch('conversations/' + S.activeChat + '/messages', 'POST', { content: txt, type: 'text' });
      if (!data.success) { msgDiv.querySelector('.msg-time').textContent = now + ' ✓'; return; }
      await loadMessages(S.activeChat, true);
    }

        renderBars(DEFAULT_BARS);
        return;
      }
      const data = await apiFetch('conversations/' + S.activeChat + '/compatibility');
      if (data.success && data.data) {
        const score = data.data.score || data.data.compatibility || '—';
        document.getElementById('reportScore').textContent = score + (typeof score === 'number' ? '%' : '');
        const bars = data.data.dimensions || DEFAULT_BARS;
        renderBars(bars);
      } else {
        document.getElementById('reportScore').textContent = '—';
        renderBars(DEFAULT_BARS);
      }
    }

    function toast(msg, type = 'info') {
      const icons = { success: '✅', error: '❌', info: '💡', warning: '⚠️' };
      const t = document.createElement('div'); t.className = 'toast';
      t.innerHTML = `<span class="toast-icon">${icons[type] || '💡'}</span><span>${msg}</span>`;
      document.getElementById('toastStack').appendChild(t);
      setTimeout(() => t.remove(), 3300);
    }

    // ══════════ SERVER NOTIFICATIONS POLLING ══════════
    let _notifPollInterval = null;
    let _lastNotifTime = null;

    async function pollServerNotifications() {
      if (!S.user) return;
      const data = await apiFetch('notifications');
      if (!data.success || !data.data) return;
      const notifs = data.data;
      for (const n of notifs) {
        if (!n.is_read && (!_lastNotifTime || new Date(n.created_at) > new Date(_lastNotifTime))) {
          if (n.type === 'match') {
            const matchName = n.title.replace("It's a Match! You and ", "").replace(/ liked each other 🎉.*/, '');
            addNotif({ icon: '🎉', text: `<b>It's a Match!</b> You and <b>${matchName}</b> liked each other!`, time: 'just now', unread: true, userId: n.actor_id });
            if (document.getElementById('matchModal').style.display !== 'flex') {
              showMatchModal(n.actor_id, matchName);
            }
          } else if (n.type === 'like') {
            const likerName = n.title.replace(' liked your profile!', '');
            addNotif({ icon: '💕', text: `<b>${likerName}</b> liked your profile!`, time: 'just now', unread: true, userId: n.actor_id });
          }
        }
      }
      if (notifs.length) _lastNotifTime = notifs[0].created_at;
      updateBadges();
    }

    function startNotifPolling() {
      if (_notifPollInterval) clearInterval(_notifPollInterval);
      _lastNotifTime = new Date().toISOString();
      _notifPollInterval = setInterval(pollServerNotifications, 8000);
    }
